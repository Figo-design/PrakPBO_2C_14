    private JPanel createPage() {
        JPanel page = new JPanel(new GridBagLayout()) {
            @Override
            protected void paintComponent(Graphics graphics) {
                super.paintComponent(graphics);
                Graphics2D g = (Graphics2D) graphics.create();
                g.setRenderingHint(RenderingHints.KEY_ANTIALIASING,
                        RenderingHints.VALUE_ANTIALIAS_ON);
                g.setColor(new Color(247, 246, 243));
                g.fillRect(0, 0, getWidth(), getHeight());
                g.dispose();
            }
        };
        page.setBorder(new EmptyBorder(16, 22, 16, 22));

        JPanel card = new JPanel();
        card.setOpaque(false);
        card.setLayout(new BoxLayout(card, BoxLayout.Y_AXIS));
        card.setBorder(new EmptyBorder(36, 30, 26, 30));

        JPanel cardSurface = new RoundedPanel(Color.WHITE, 18);
        cardSurface.setLayout(new BorderLayout());
        cardSurface.add(card, BorderLayout.CENTER);

        GridBagConstraints constraints = new GridBagConstraints();
        constraints.fill = GridBagConstraints.BOTH;
        constraints.weightx = 1;
        constraints.weighty = 1;
        page.add(cardSurface, constraints);

        card.add(createHeader());
        card.add(Box.createVerticalStrut(38));

        card.add(createFieldLabel("Email"));
        card.add(Box.createVerticalStrut(8));
        card.add(createEmailField());
        card.add(createEmailHint());
        card.add(Box.createVerticalStrut(14));

        card.add(createFieldLabel("Kata Sandi"));
        card.add(Box.createVerticalStrut(8));
        card.add(createPasswordField());
        card.add(Box.createVerticalStrut(19));
        card.add(createRememberCheckbox());
        card.add(Box.createVerticalStrut(18));

        JButton loginButton = createFilledButton("↪   Masuk ke Akun");
        loginButton.addActionListener(this::handleLogin);
        card.add(loginButton);

        card.add(createForgotPasswordLink());
        card.add(Box.createVerticalStrut(4));

        JButton homeButton = createOutlineButton("Kembali ke Beranda");
        homeButton.addActionListener(event -> dispose());
        card.add(homeButton);
        card.add(Box.createVerticalStrut(24));
        card.add(createSignupPrompt());

        return page;
    }

        passwordField.setFont(new Font("SansSerif", Font.PLAIN, 17));
        passwordField.setForeground(TEXT);
        passwordField.setBorder(BorderFactory.createEmptyBorder(0, 8, 0, 8));

        button.setCursor(Cursor.getPredefinedCursor(Cursor.HAND_CURSOR));

    private void handleLogin(ActionEvent event) {
        String email = emailField.getText().trim();
        char[] password = passwordField.getPassword();
        boolean validEmail = email.matches("^[A-Za-z0-9+_.-]+@[A-Za-z0-9.-]+$");
        boolean hasPassword = password.length > 0;
        Arrays.fill(password, '\0');

        if (!validEmail || !hasPassword) {
            emailHint.setText(!validEmail
                    ? "Masukkan alamat email yang valid"
                    : "Email valid");
            emailHint.setForeground(!validEmail ? new Color(190, 53, 53) : MUTED);
            JOptionPane.showMessageDialog(this,
                    "Periksa kembali email dan kata sandi Anda.",
                    "Data belum lengkap",
                    JOptionPane.WARNING_MESSAGE);
            return;
        }

        emailHint.setText("Email valid");
        emailHint.setForeground(GREEN);
        showNotAvailableMessage("Login");
    }

    private void showNotAvailableMessage(String feature) {
        JOptionPane.showMessageDialog(this,
                feature + " belum terhubung ke sistem.",
                "Informasi",
                JOptionPane.INFORMATION_MESSAGE);
    }

    private static class RoundedPanel extends JPanel {
        private final Color color;
        private final int radius;

        RoundedPanel(Color color, int radius) {
            this.color = color;
            this.radius = radius;
            setOpaque(false);
        }

        @Override
        protected void paintComponent(Graphics graphics) {
            Graphics2D g = (Graphics2D) graphics.create();
            g.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
            g.setColor(new Color(0, 0, 0, 16));
            g.fillRoundRect(1, 3, getWidth() - 2, getHeight() - 4, radius, radius);
            g.setColor(color);
            g.fillRoundRect(0, 0, getWidth() - 2, getHeight() - 5, radius, radius);
            g.dispose();
            super.paintComponent(graphics);
        }
    }

    private static class RoundedBorder extends AbstractBorder {
        private final Color color;
        private final int radius;

        RoundedBorder(Color color, int radius) {
            this.color = color;
            this.radius = radius;
        }

        @Override
        public Insets getBorderInsets(Component component) {
            return new Insets(1, 1, 1, 1);
        }

        @Override
        public void paintBorder(Component component, Graphics graphics,
                int x, int y, int width, int height) {
            Graphics2D g = (Graphics2D) graphics.create();
            g.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
            g.setColor(color);
            g.setStroke(new BasicStroke(1f));
            g.drawRoundRect(x, y, width - 1, height - 1, radius, radius);
            g.dispose();
        }
    }

    private abstract static class LineIcon implements Icon {
        private final int size;

        LineIcon(int size) {
            this.size = size;
        }

        @Override
        public int getIconWidth() {
            return size;
        }

        @Override
        public int getIconHeight() {
            return size;
        }

        protected Graphics2D prepare(Graphics graphics) {
            Graphics2D g = (Graphics2D) graphics.create();
            g.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
            g.setColor(MUTED);
            g.setStroke(new BasicStroke(1.7f, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
            return g;
        }
    }

    private static class MailIcon extends LineIcon {

            g.drawLine(x + 3, y + 5, x + 10, y + 11);
            g.drawLine(x + 17, y + 5, x + 10, y + 11);
            g.dispose();
        }
    }

    private static class HouseIcon extends LineIcon {
        HouseIcon() {
            super(54);
        }

        @Override
        public void paintIcon(Component component, Graphics graphics, int x, int y) {
            Graphics2D g = (Graphics2D) graphics.create();
            g.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
            g.setStroke(new BasicStroke(2.2f, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
            g.setColor(new Color(34, 35, 33));
            int[] roofX = {x + 8, x + 27, x + 46};
            int[] roofY = {y + 22, y + 3, y + 22};
            g.drawPolyline(roofX, roofY, 3);
            g.setColor(new Color(113, 191, 55));
            int[] fillX = {x + 13, x + 27, x + 41};
            int[] fillY = {y + 19, y + 5, y + 19};
            g.fillPolygon(fillX, fillY, 3);
            g.setColor(new Color(34, 35, 33));
            g.drawLine(x + 15, y + 19, x + 15, y + 41);
            g.drawLine(x + 39, y + 19, x + 39, y + 41);
            g.drawLine(x + 15, y + 41, x + 39, y + 41);
            g.setColor(new Color(242, 170, 45));
            g.fillRect(x + 17, y + 21, 20, 18);
            g.setColor(new Color(34, 35, 33));
            g.drawRect(x + 17, y + 21, 20, 18);
            g.drawLine(x + 27, y + 22, x + 27, y + 38);
            g.drawLine(x + 24, y + 29, x + 30, y + 29);
            g.drawLine(x + 5, y + 43, x + 49, y + 43);
            g.dispose();
        }
